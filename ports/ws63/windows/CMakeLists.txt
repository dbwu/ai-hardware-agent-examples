cmake_minimum_required(VERSION 3.21)
project(unified_ai_ws63_simulator C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 14)
get_filename_component(app_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CMAKE_CURRENT_LIST_DIR}/../board.cmake")
set(CONVAI_SDK_ROOT "${app_root}/third_party/convai-sdk-26.8.0" CACHE PATH "ConvAI SDK distribution")
set(WS63_LOCAL_CONFIG "${app_root}/config/ws63.local.h" CACHE FILEPATH "Private credentials shared with board")
set(sdk_include "${CONVAI_SDK_ROOT}/include")
get_filename_component(examples_root "${GOLDIEOS_ROOT}/../.." ABSOLUTE)
include("${examples_root}/cmake/convai-build-options.cmake")
include("${examples_root}/cmake/convai-artifacts.cmake")
include("${GOLDIEOS_ROOT}/cmake/goldieos-common.cmake")
include("${GOLDIEOS_ROOT}/cmake/goldieos-win.cmake")
add_library(convai_sdk STATIC IMPORTED GLOBAL)
set_target_properties(convai_sdk PROPERTIES
    IMPORTED_LOCATION "${CONVAI_SDK_ROOT}/libs/windows/libconvai_sdk.a"
    INTERFACE_INCLUDE_DIRECTORIES "${sdk_include};${sdk_include}/convai")
goldieos_add_opus(PLATFORM goldieos SOURCE_DIR "${GOLDIEOS_ROOT}")
goldieos_add_win_target(TARGET unified_ai_simulator SOURCE_DIR "${GOLDIEOS_ROOT}" VERSION "0.1.0")
get_target_property(sources unified_ai_simulator SOURCES)
list(FILTER sources EXCLUDE REGEX "/apps/|/sdk_integration/|/services/aud_algo/")
list(FILTER sources EXCLUDE REGEX "/init/system_init\\.c$|/platform/convai_platform_win\\.c$")
set_property(TARGET unified_ai_simulator PROPERTY SOURCES "${sources}")
include("${CMAKE_CURRENT_LIST_DIR}/../common.cmake")
unified_ws63_common(unified_ai_simulator)
target_sources(unified_ai_simulator PRIVATE runtime.c sdk_platform.c)
target_compile_definitions(unified_ai_simulator PRIVATE AI_UI_NO_PROVISION)
target_link_libraries(unified_ai_simulator PRIVATE wininet)
set_target_properties(unified_ai_simulator PROPERTIES LINKER_LANGUAGE CXX)
